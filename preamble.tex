%% This file contains a number of tweaks that are typically applied to the main document.
%% They are not enabled by default, but can be enabled by uncommenting the relevant lines.

% Backport the lineno 5.9 two-column fix for TeX Live 2025's lineno 5.6--5.8.
% These versions write column markers in the wrong place, putting review
% line numbers over the opposite column. Newer lineno needs no patch.
% https://github.com/latex-lineno/lineno/issues/26
\makeatletter
\@ifpackageloaded{lineno}{%
  \@ifpackagelater{lineno}{2026/02/10}{%
    \@ifpackagelater{lineno}{2026/03/07}{}{%
      \IfFormatAtLeastTF{2025-06-01}{%
        \RemoveFromHook{build/column/before}[lineno]
        \IfHookEmptyTF{build/column/after}{}{%
          \RemoveFromHook{build/column/after}[lineno]%
        }%
        \AddToHook{build/column/after}[cvpr-lineno-fix]{%
          \setbox\@outputbox\vbox to\ht\@outputbox{%
            \boxmaxdepth\@maxdepth
            \protected@write\@auxout{}{%
              \string\def\string\@LN@column{\if@firstcolumn1\else2\fi}%
            }%
            \unvbox\@outputbox
          }%
        }%
      }{}%
    }%
  }{}%
}{}%
\makeatother
